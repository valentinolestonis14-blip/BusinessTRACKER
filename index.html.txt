<script>
// ================= INTERACTIVE CAROUSEL BUSINESS BACKGROUND IMPLEMENTATION =================
const bgSliderNode = document.getElementById("premiumCarouselBg");
const carouselImages = JSON.parse(bgSliderNode.getAttribute("data-bg-carousel"));
let currentBgIndex = 0;

function updateCarouselUI() {
    // Palitan ang Active Engine Image
    bgSliderNode.style.backgroundImage = `url('${carouselImages[currentBgIndex]}')`;
    
    // Refresh Navigation Indicator Dots
    const dotsContainer = document.getElementById("carouselIndicatorDots");
    dotsContainer.innerHTML = "";
    carouselImages.forEach((_, index) => {
        const dot = document.createElement("div");
        dot.className = `h-2 rounded-full transition-all duration-300 ${index === currentBgIndex ? 'w-5 bg-blue-500' : 'w-2 bg-white/40'}`;
        dotsContainer.appendChild(dot);
    });
}

function navigateBg(direction) {
    currentBgIndex += direction;
    if (currentBgIndex >= carouselImages.length) currentBgIndex = 0;
    if (currentBgIndex < 0) currentBgIndex = carouselImages.length - 1;
    updateCarouselUI();
}

// 1. Keyboard Shortcuts (Left/Right Keys)
window.addEventListener("keydown", (e) => {
    if (e.key === "ArrowLeft") navigateBg(-1);
    if (e.key === "ArrowRight") navigateBg(1);
});

// 2. Swipable & Drag Gestures (Mobile/Mouse Drag Events)
let touchStartX = 0;
let touchEndX = 0;

window.addEventListener("touchstart", e => touchStartX = e.changedTouches[0].screenX, {passive: true});
window.addEventListener("touchend", e => {
    touchEndX = e.changedTouches[0].screenX;
    handleSwipeGesture();
}, {passive: true});

function handleSwipeGesture() {
    if (touchStartX - touchEndX > 60) navigateBg(1);  // Swipe Left
    if (touchEndX - touchStartX > 60) navigateBg(-1); // Swipe Right
}

// Initialize Dynamic Framework
updateCarouselUI();


// ================= YOUR MAIN CORE ORIGINAL LOGIC ENGINE =================
let products = JSON.parse(localStorage.getItem("products")) || [];
let sales = JSON.parse(localStorage.getItem("sales")) || [];
let orders = JSON.parse(localStorage.getItem("orders")) || [];

function saveData(){
    localStorage.setItem("products", JSON.stringify(products));
    localStorage.setItem("sales", JSON.stringify(sales));
    localStorage.setItem("orders", JSON.stringify(orders));
}

function showSection(section){

    document.querySelectorAll("main > section").forEach(sec=>{
        sec.classList.add("hidden");
    });

    document.getElementById(section + "Section").classList.remove("hidden");
}

function toggleSidebar(){
    const sidebar = document.getElementById("sidebar");
    if(sidebar.classList.contains("-translate-x-full")) {
        sidebar.classList.remove("-translate-x-full");
    } else {
        sidebar.classList.add("-translate-x-full");
    }
}

function openProductModal(){
    document.getElementById("productModal").classList.add("active");
}

function closeProductModal(){
    document.getElementById("productModal").classList.remove("active");
}

function openOrderModal(){
    document.getElementById("orderModal").classList.add("active");
}

function closeOrderModal(){
    document.getElementById("orderModal").classList.remove("active");
}

function addProduct(){

    let product = {
        id: Date.now(),
        name: document.getElementById("productName").value,
        price: document.getElementById("productPrice").value,
        category: document.getElementById("productCategory").value,
        stock: document.getElementById("productStock").value,
        image: document.getElementById("productImage").value,
        description: document.getElementById("productDescription").value
    };

    products.push(product);

    saveData();

    renderProducts();
    updateDashboard();
    renderInventory();

    closeProductModal();
}

function renderProducts(){

    let searchInput = document.getElementById("searchProduct");
    let search = searchInput ? searchInput.value.toLowerCase() : "";

    let html = "";

    products
    .filter(p=>p.name.toLowerCase().includes(search))
    .forEach(product=>{

        html += `
        <div class="glass rounded-3xl overflow-hidden card-hover">
            <img src="${product.image}" class="w-full h-56 object-cover" onerror="this.src='https://images.unsplash.com/photo-1531403009284-440f080d1e12?auto=format&fit=crop&w=600&q=80'">

            <div class="p-6">
                <div class="flex justify-between items-center mb-3">

function addSale(){

    let quantity = Number(document.getElementById("saleQuantity").value);
    let price = Number(document.getElementById("salePrice").value);

    let total = quantity * price;

    let selectedDate = document.getElementById("saleDate").value;

    if(!selectedDate){
        alert("Please select a date");
        return;
    }

    let dayName = new Date(selectedDate).toLocaleDateString('en-US', {
        weekday: 'long'
    });

    let sale = {
        id: Date.now(),
        name: document.getElementById("saleName").value,
        quantity: quantity,
        price: price,
        amount: total,
        day: dayName,
        date: selectedDate,
        type: document.getElementById("saleType").value
    };

    sales.push(sale);

    saveData();

    renderSalesTable();
    updateCharts();
    updateDashboard();

    document.getElementById("saleName").value = "";
    document.getElementById("saleQuantity").value = "";
    document.getElementById("salePrice").value = "";
    document.getElementById("saleDate").value = "";
}

function updateCharts(){

    if(salesChart) salesChart.destroy();
    if(productChart) productChart.destroy();
    if(dailyChart) dailyChart.destroy();
    if(weeklyChart) weeklyChart.destroy();
    if(monthlyChart) monthlyChart.destroy();
    if(reportChart) reportChart.destroy();

    let daily = sales.filter(s=>s.type=="Daily");
    let weekly = sales.filter(s=>s.type=="Weekly");
    let monthly = sales.filter(s=>s.type=="Monthly");

    salesChart = createChart(
        "salesChart",
        sales.map(s=>s.name),
        sales.map(s=>s.amount),
        "Sales"
    );

    productChart = new Chart(document.getElementById("productChart"),{
        type:'pie',

        data:{
            labels:products.map(p=>p.name),

            datasets:[{
                data:products.map(p=>p.stock),
                borderWidth: 0,
                backgroundColor:[
                    "#3b82f6",
                    "#8b5cf6",
                    "#ec4899",
                    "#10b981",
                    "#f59e0b"
                ]
            }]
        },
        options: {
            responsive: true,
            plugins: {
                legend: {
                    labels: {
                        color: 'white',
                        font: { family: "'Inter', sans-serif" }
                    }
                }
            }
        }
    });

    dailyChart = createChart(
        "dailyChart",
        daily.map(s=>s.name),
        daily.map(s=>s.amount),
        "Daily"
    );

    weeklyChart = createChart(
        "weeklyChart",
        weekly.map(s=>s.name),
        weekly.map(s=>s.amount),
        "Weekly"
    );

    monthlyChart = createChart(
        "monthlyChart",
        monthly.map(s=>s.name),
        monthly.map(s=>s.amount),
        "Monthly"
    );

    reportChart = createChart(
        "reportChart",
        sales.map(s=>s.name),
        sales.map(s=>s.amount),
        "Reports"
    );
}

const bgSliderNode = document.getElementById("premiumCarouselBg");
const carouselImages = JSON.parse(bgSliderNode.getAttribute("data-bg-carousel"));
let currentBgIndex = 0;

function updateCarouselUI() {
    // Palitan ang Active Engine Image
    bgSliderNode.style.backgroundImage = `url('${carouselImages[currentBgIndex]}')`;
    
    // Refresh Navigation Indicator Dots
    const dotsContainer = document.getElementById("carouselIndicatorDots");
    dotsContainer.innerHTML = "";
    carouselImages.forEach((_, index) => {
        const dot = document.createElement("div");
        dot.className = `h-2 rounded-full transition-all duration-300 ${index === currentBgIndex ? 'w-5 bg-blue-500' : 'w-2 bg-white/40'}`;
        dotsContainer.appendChild(dot);
    });
}

function navigateBg(direction) {
    currentBgIndex += direction;
    if (currentBgIndex >= carouselImages.length) currentBgIndex = 0;
    if (currentBgIndex < 0) currentBgIndex = carouselImages.length - 1;
    updateCarouselUI();
}

// 1. Keyboard Shortcuts (Left/Right Keys)
window.addEventListener("keydown", (e) => {
    if (e.key === "ArrowLeft") navigateBg(-1);
    if (e.key === "ArrowRight") navigateBg(1);
});

// 2. Swipable & Drag Gestures (Mobile/Mouse Drag Events)
let touchStartX = 0;
let touchEndX = 0;

window.addEventListener("touchstart", e => touchStartX = e.changedTouches[0].screenX, {passive: true});
window.addEventListener("touchend", e => {
    touchEndX = e.changedTouches[0].screenX;
    handleSwipeGesture();
}, {passive: true});

function handleSwipeGesture() {
    if (touchStartX - touchEndX > 60) navigateBg(1);  // Swipe Left
    if (touchEndX - touchStartX > 60) navigateBg(-1); // Swipe Right
}

// Initialize Dynamic Framework
updateCarouselUI();
