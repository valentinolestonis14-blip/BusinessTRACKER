<script>
let products = JSON.parse(localStorage.getItem("products")) || [];
let sales = JSON.parse(localStorage.getItem("sales")) || [];
let orders = JSON.parse(localStorage.getItem("orders")) || [];

function saveData(){
    localStorage.setItem("products", JSON.stringify(products));
    localStorage.setItem("sales", JSON.stringify(sales));
    localStorage.setItem("orders", JSON.stringify(orders));
}

function showSection(section){

    document.querySelectorAll("section").forEach(sec=>{
        sec.classList.add("hidden");
    });

    document.getElementById(section + "Section").classList.remove("hidden");
}

function toggleSidebar(){
    document.getElementById("sidebar").classList.toggle("-translate-x-full");
}

function openProductModal(){
    document.getElementById("productModal").classList.add("active");
}

function closeProductModal(){
    document.getElementById("productModal").classList.remove("active");
}

function openOrderModal(){
    document.getElementById("orderModal").classList.add("active");
}

function closeOrderModal(){
    document.getElementById("orderModal").classList.remove("active");
}

function addProduct(){

    let product = {
        id: Date.now(),
        name: document.getElementById("productName").value,
        price: document.getElementById("productPrice").value,
        category: document.getElementById("productCategory").value,
        stock: document.getElementById("productStock").value,
        image: document.getElementById("productImage").value,
        description: document.getElementById("productDescription").value
    };

    products.push(product);

    saveData();

    renderProducts();
    updateDashboard();
    renderInventory();

    closeProductModal();
}

function renderProducts(){

    let search = document.getElementById("searchProduct").value.toLowerCase();

    let html = "";

    products
    .filter(p=>p.name.toLowerCase().includes(search))
    .forEach(product=>{

        html += `
        <div class="glass rounded-3xl overflow-hidden card-hover">
            <img src="${product.image}" class="w-full h-56 object-cover">

            <div class="p-6">
                <div class="flex justify-between items-center mb-3">
                    <h2 class="text-2xl font-bold">${product.name}</h2>
                    <span class="text-blue-400 font-bold">₱${product.price}</span>
                </div>

                <p class="text-gray-400 mb-4">${product.description}</p>

                <div class="flex justify-between items-center">
                    <span class="glass px-4 py-2 rounded-xl">
                        ${product.stock} Stocks
                    </span>

                    <button onclick="deleteProduct(${product.id})"
                        class="bg-red-500 hover:bg-red-600 px-4 py-2 rounded-xl">
                        Delete
                    </button>
                </div>
            </div>
        </div>
        `;
    });

    document.getElementById("productList").innerHTML = html;
}

function deleteProduct(id){

    products = products.filter(p=>p.id !== id);

    saveData();

    renderProducts();
    updateDashboard();
    renderInventory();
}

function addSale(){

    let quantity = Number(document.getElementById("saleQuantity").value);
    let price = Number(document.getElementById("salePrice").value);

    let total = quantity * price;

    let selectedDate = document.getElementById("saleDate").value;

    if(!selectedDate){
        alert("Please select a date");
        return;
    }

    let dayName = new Date(selectedDate).toLocaleDateString('en-US', {
        weekday: 'long'
    });

    let sale = {
        id: Date.now(),
        name: document.getElementById("saleName").value,
        quantity: quantity,
        price: price,
        amount: total,
        day: dayName,
        date: selectedDate,
        type: document.getElementById("saleType").value
    };

    sales.push(sale);

    saveData();

    renderSalesTable();
    updateCharts();
    updateDashboard();

    document.getElementById("saleName").value = "";
    document.getElementById("saleQuantity").value = "";
    document.getElementById("salePrice").value = "";
    document.getElementById("saleDate").value = "";
}
function renderSalesTable(){

    let html = "";

    sales.forEach(sale => {

        html += `
        <tr class="border-b border-white/10">

            <td class="p-4">${sale.name}</td>

            <td class="p-4">${sale.quantity}</td>

            <td class="p-4">₱${sale.price}</td>

            <td class="p-4 text-green-400 font-bold">
                ₱${sale.amount}
            </td>

            <td class="p-4">${sale.day}</td>

            <td class="p-4">${sale.date}</td>

            <td class="p-4">
                <span class="glass px-4 py-2 rounded-xl">
                    ${sale.type}
                </span>
            </td>

            <td class="p-4">

                <button onclick="deleteSale(${sale.id})"
                    class="bg-red-500 hover:bg-red-600 px-4 py-2 rounded-xl">
                    Delete
                </button>

            </td>

        </tr>
        `;
    });

    document.getElementById("salesTable").innerHTML = html;
}

function deleteSale(id){

    sales = sales.filter(s => s.id !== id);

    saveData();

    renderSalesTable();
    updateCharts();
    updateDashboard();
}
function addOrder(){

    let order = {
        id: Date.now(),
        customer: document.getElementById("customerName").value,
        product: document.getElementById("orderProduct").value,
        quantity: document.getElementById("orderQuantity").value,
        status: document.getElementById("orderStatus").value
    };

    orders.push(order);

    saveData();

    renderOrders();
    updateDashboard();

    closeOrderModal();
}

function renderOrders(){

    let html = "";

    orders.forEach(order=>{

        html += `
        <div class="glass rounded-3xl p-6 flex flex-col md:flex-row justify-between gap-4">

            <div>
                <h2 class="text-2xl font-bold">
                    ${order.customer}
                </h2>

                <p class="text-gray-400">
                    ${order.product} x ${order.quantity}
                </p>
            </div>

            <div class="flex gap-4 items-center">

                <span class="glass px-4 py-2 rounded-xl">
                    ${order.status}
                </span>

                <button onclick="deleteOrder(${order.id})"
                    class="bg-red-500 hover:bg-red-600 px-4 py-2 rounded-xl">
                    Delete
                </button>

            </div>

        </div>
        `;
    });

    document.getElementById("orderList").innerHTML = html;
}

function deleteOrder(id){

    orders = orders.filter(o=>o.id !== id);

    saveData();

    renderOrders();
    updateDashboard();
}

function renderInventory(){

    let html = "";

    products.forEach(product=>{

        html += `
        <tr class="border-b border-white/10">
            <td class="p-4">${product.name}</td>

            <td class="p-4">${product.stock}</td>

            <td class="p-4">
                ${product.stock <= 5 ?
                    '<span class="text-red-400">Low Stock</span>'
                    :
                    '<span class="text-green-400">In Stock</span>'
                }
            </td>
        </tr>
        `;
    });

    document.getElementById("inventoryTable").innerHTML = html;
}

function updateDashboard(){

    let revenue = sales.reduce((a,b)=> a + Number(b.amount), 0);

    document.getElementById("totalRevenue").innerHTML = "₱" + revenue;

    document.getElementById("totalProducts").innerHTML = products.length;

    document.getElementById("totalOrders").innerHTML = orders.length;

    let low = products.filter(p=>p.stock <=5).length;

    document.getElementById("lowStocks").innerHTML = low;
}

function resetProducts(){

    if(confirm("Reset all products?")){

        products = [];

        saveData();

        renderProducts();
        renderInventory();
        updateDashboard();
    }
}

function resetOrders(){

    if(confirm("Reset all orders?")){

        orders = [];

        saveData();

        renderOrders();
        updateDashboard();
    }
}

function resetInventory(){

    if(confirm("Reset inventory?")){

        products.forEach(p=>p.stock = 0);

        saveData();

        renderInventory();
        updateDashboard();
    }
}

function resetReports(){

    if(confirm("Reset reports?")){

        sales = [];

        saveData();

        updateCharts();
        updateDashboard();
    }
}

function resetSales(){

    if(confirm("Reset all sales?")){

        sales = [];

        saveData();

        updateCharts();
        updateDashboard();
    }
}

function resetType(type){

    sales = sales.filter(s=>s.type !== type);

    saveData();

    updateCharts();
    updateDashboard();
}

function downloadJSON(){

    const data = {
        products,
        sales,
        orders
    };

    const blob = new Blob([JSON.stringify(data,null,2)],{
        type:"application/json"
    });

    const a = document.createElement("a");

    a.href = URL.createObjectURL(blob);

    a.download = "business-report.json";

function downloadCSV(){

    let csv = "Type,Name,Amount\n";

    sales.forEach(s=>{
        csv += `${s.type},${s.name},${s.amount}\n`;
    });

    const blob = new Blob([csv],{
        type:"text/csv"
    });

    const a = document.createElement("a");

    a.href = URL.createObjectURL(blob);

    a.download = "sales-report.csv";

    a.click();
}
